local BASE = ("https://raw.githubusercontent.com/%s/%s/%s/"):format(CONFIG.OWNER, CONFIG.REPO, CONFIG.BRANCH)

local function httpGet(url)
    local ok, res = pcall(function()
        return game:HttpGet(url)
    end)
    if ok and type(res) == "string" and #res > 0 then return res end

    local req = (syn and syn.request) or http_request or request or (http and http.request)
    if req then
        local ok2, r = pcall(req, { Url = url, Method = "GET" })
        if ok2 and type(r) == "table" and type(r.Body) == "string" and #r.Body > 0
            and (r.StatusCode == nil or r.StatusCode == 200) then
            return r.Body
        end
    end
    return nil
end

-- Devuelve el asset listo para Roblox, o nil + el motivo del error.
local function fetchFile(entry)
    if not (writefile and getAsset) then
        return nil, "tu ejecutor no soporta writefile/getcustomasset"
    end
    local cache = CONFIG.CACHE_PREFIX .. entry.file

    -- 1) ¿ya está guardado y completo? -> carga instantánea
    if isfile and readfile then
        local okI, exists = pcall(isfile, cache)
        if okI and exists then
            local okR, data = pcall(readfile, cache)
            if okR and type(data) == "string" and #data == entry.size then
                local okA, asset = pcall(getAsset, cache)
                if okA and asset then return asset end
            end
        end
    end

    -- 2) descargar (hasta 3 intentos)
    local err = entry.file .. ": no se pudo descargar de GitHub"
    for attempt = 1, 3 do
        local data = httpGet(BASE .. entry.file)
        if data and (#data == entry.size or (attempt == 3 and #data > 1000)) then
            local okW, errW = pcall(writefile, cache, data)
            if not okW then return nil, entry.file .. ": writefile falló (" .. tostring(errW) .. ")" end
            local okA, asset = pcall(getAsset, cache)
            if okA and asset then return asset end
            return nil, entry.file .. ": getcustomasset falló"
        end
        if data then
            err = ("%s: llegó incompleto (%d de %d bytes)"):format(entry.file, #data, entry.size)
        end
        task.wait(0.3 * attempt)
    end
    return nil, err
end

-- ================= ESTADO =================
local FRAMES_PER_SHEET = CONFIG.COLS * CONFIG.ROWS
local DURATION = CONFIG.FRAMES / CONFIG.FPS
local TOTAL_JOBS = 1 + #CONFIG.SHEETS

local sheets = {}          -- [n] = { label = ImageLabel, ready = bool }
local activeSheet = nil
local shownFrame = -1
local sound = nil
local audioState = "loading" -- "loading" | "ready" | "failed"
local playing = false
local isOpen = false
local clock = 0            -- reloj interno (solo se usa si la música no cargó)
local lastT = 0
local volume = CONFIG.START_VOLUME
local lastVolume = CONFIG.START_VOLUME
local muted = false
local doneJobs = 0
local failures = {}

local conns = {}
local function connect(signal, fn)
    local c = signal:Connect(fn)
    conns[#conns + 1] = c
    return c
end

gui.Destroying:Connect(function()
    for _, c in ipairs(conns) do
        pcall(function() c:Disconnect() end)
    end
end)

local function refreshStatus()
    if #failures > 0 then
        status.Text = "No cargó: " .. failures[1]
        status.TextColor3 = Color3.fromRGB(255, 130, 130)
    elseif doneJobs < TOTAL_JOBS then
        status.Text = ("Cargando %d/%d..."):format(doneJobs, TOTAL_JOBS)
        status.TextColor3 = Color3.fromRGB(255, 220, 120)
    else
        status.Text = ""
    end
end

-- ================= VOLUMEN =================
local function setVolume(v)
    volume = math.clamp(v, 0, 1)
    sliderFill.Size = UDim2.fromScale(volume, 1)
    knob.Position = UDim2.new(volume, 0, 0.5, 0)
    if volume > 0 then
        lastVolume = volume
        muted = false
        muteBtn.Text = "Sonido"
    else
        muted = true
        muteBtn.Text = "Mute"
    end
    if sound then sound.Volume = volume end
end

-- ================= REPRODUCCIÓN =================
local function startPlayback(fromStart)
    playing = true
    if fromStart then
        lastT = 0
        clock = 0
        shownFrame = -1
    end
    if sound then
        if fromStart then sound.TimePosition = 0 end
        sound:Play()
    end
    playBtn.Text = "Pausar"
end

local function pausePlayback()
    playing = false
    if sound then sound:Pause() end
    playBtn.Text = "Reanudar"
end

-- Muestra el fotograma f (0 .. FRAMES-1). Devuelve false si su hoja aún no llegó.
local function showFrame(f)
    local sheet = sheets[math.floor(f / FRAMES_PER_SHEET) + 1]
    if not (sheet and sheet.ready) then return false end

    local k = f % FRAMES_PER_SHEET
    local col = k % CONFIG.COLS
    local row = math.floor(k / CONFIG.COLS)
    -- +1 / -2: recorta 1 px por lado para no mostrar la línea negra entre fotogramas
    local x = CONFIG.MARGIN + col * (CONFIG.FRAME_W + CONFIG.PADDING) + 1
    local y = CONFIG.MARGIN + row * (CONFIG.FRAME_H + CONFIG.PADDING) + 1

    local label = sheet.label
    label.ImageRectOffset = Vector2.new(x, y)
    label.ImageRectSize = Vector2.new(CONFIG.FRAME_W - 2, CONFIG.FRAME_H - 2)

    if activeSheet ~= sheet then
        if activeSheet then activeSheet.label.ZIndex = 1 end
        label.ZIndex = 3 -- por encima de la tapa negra
        activeSheet = sheet
    end
    if loadingLabel.Visible then loadingLabel.Visible = false end
    return true
end

-- La música manda: el fotograma sale de la posición de la música.
local function currentTime(dt)
    if sound then
        if sound.IsPlaying then lastT = sound.TimePosition end
        return lastT
    end
    if audioState == "failed" and playing then
        clock = clock + dt
        if clock >= DURATION then clock = clock - DURATION end
        lastT = clock
    end
    return lastT
end

local function update(dt)
    if not isOpen then return end
    local t = currentTime(dt)
    local f = math.floor(t * CONFIG.FPS)
    if f < 0 then f = 0 end
    if f > CONFIG.FRAMES - 1 then f = CONFIG.FRAMES - 1 end
    if f ~= shownFrame then
        if showFrame(f) then shownFrame = f end
    end
end

local updateError = false
connect(RunService.Heartbeat, function(dt)
    local ok, err = pcall(update, dt)
    if not ok and not updateError then
        updateError = true
        status.Text = "Error: " .. tostring(err)
    end
end)

-- ================= CARGA: primero la música, luego las imágenes en orden =================
local function addSheet(n, asset)
    local fw, fh = CONFIG.FRAME_W - 2, CONFIG.FRAME_H - 2
    local label = make("ImageLabel", {
        Name = "Sheet" .. n,
        BackgroundTransparency = 1,
        Size = UDim2.fromScale(1, 1),
        Image = asset,
        -- ya apunta al primer fotograma de la hoja (nunca se ve la hoja entera)
        ImageRectOffset = Vector2.new(CONFIG.MARGIN + 1, CONFIG.MARGIN + 1),
        ImageRectSize = Vector2.new(fw, fh),
        ZIndex = 1,
    }, videoHolder)

    local sheet = { label = label, ready = false }
    sheets[n] = sheet

    -- espera a que Roblox termine de cargar la imagen (máx. 5 s) y la marca lista
    task.spawn(function()
        local t0 = os.clock()
        while os.clock() - t0 < 5 do
            local ok, loaded = pcall(function() return label.IsLoaded end)
            if not ok then
                task.wait(0.25)
                break
            end
            if loaded then break end
            task.wait()
        end
        sheet.ready = true
    end)
end

local function addAudio(asset)
    local s = make("Sound", {
        Name = "MenuAudio",
        SoundId = asset,
        Looped = true,
        Volume = volume,
    }, gui)
    sound = s
    audioState = "ready"
    if playing and isOpen then
        s.TimePosition = 0
        s:Play() -- ¡suena apenas llega!
    end
end

local function startLoading()
    local jobs = {}

    jobs[#jobs + 1] = function()
        local asset, err = fetchFile(CONFIG.AUDIO)
        if asset then
            addAudio(asset)
        else
            audioState = "failed" -- el video avanza solo, sin música
            failures[#failures + 1] = err or CONFIG.AUDIO.file
        end
    end

    for n, entry in ipairs(CONFIG.SHEETS) do
        jobs[#jobs + 1] = function()
            local asset, err = fetchFile(entry)
            if asset then
                addSheet(n, asset)
            else
                failures[#failures + 1] = err or entry.file
                if n == 1 then loadingLabel.Text = "No se pudo cargar el video" end
            end
        end
    end

    local nextJob = 1
    local function worker()
        while true do
            local j = nextJob
            if j > #jobs then return end
            nextJob = j + 1
            local ok, err = pcall(jobs[j])
            if not ok then failures[#failures + 1] = tostring(err) end
            doneJobs = doneJobs + 1
            refreshStatus()
        end
    end

    refreshStatus()
    for _ = 1, CONFIG.PARALLEL do
        task.spawn(worker)
    end
end

    isOpen = true

    openBtn.Visible = false
    startPlayback(true) -- entras al menú: video y música desde el principio
end

    if not isOpen then return end
    isOpen = false
    main.Visible = false
    openBtn.Visible = true
    pausePlayback()
end

        or input.UserInputType == Enum.UserInputType.Touch then
        draggingWindow = true

        startPos = main.Position
    end

    if input.UserInputType ~= Enum.UserInputType.MouseMovement
        and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end
    if draggingSlider then
        updateSlider(input.Position.X)
    elseif draggingWindow then
        local d = input.Position - dragStart
        main.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + d.X,
            startPos.Y.Scale, startPos.Y.Offset + d.Y
        )
    end
end)

    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        draggingSlider = false
        draggingWindow = false
    end
end)
